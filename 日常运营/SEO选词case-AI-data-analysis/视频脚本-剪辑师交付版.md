### A.1 · 录制 R-1 · Phase 1 联网种子词探查

```bash
#!/bin/bash
# 录制 R-1 · InfiniSynapse 联网搜索 60 个种子词
# 执行前确认: 1) Phase 1 prompt 已经写在 prompts/phase1-联网种子词探查.md
#            2) (可选) 已经预跑过一次, taskId 记在 raw/phase1-taskId.txt
# 录屏软件 (OBS) 先开始录制再执行本脚本

AGENT=~/.auto-coder/.autocodertools/agent-browser
set -e

# ============ §1 登录态预检 ============
$AGENT daemon status || $AGENT daemon start
$AGENT set viewport 1920 1080
$AGENT open "https://app.infinisynapse.com/tasks"
$AGENT wait 2500

URL=$($AGENT get url)
[[ "$URL" == *"/login"* ]] && { echo "未登录,请手动扫码"; exit 1; }
echo "[OK] 登录态正常: $URL"

# ============ §3 直接 fill Phase 1 prompt 到首页输入框 ============
# 首页输入框是 ProseMirror, 走 SKILL.md §3.1 的注入方式
$AGENT wait 1500   # 让 OBS 录屏先稳定 1.5s 录个静止首屏

PROMPT=$(cat prompts/phase1-联网种子词探查.md | sed -n '/^```text/,/^```$/p' | sed '1d;$d')

$AGENT eval "(() => {
  const ed = document.querySelector('.ProseMirror');
  if (!ed) return 'no-editor';
  ed.focus();
  // 用 execCommand 注入(SKILL.md §3.1 实测最稳)
  document.execCommand('insertText', false, $(echo "$PROMPT" | python3 -c 'import sys, json; print(json.dumps(sys.stdin.read()))'));
  return 'ok';
})()"
$AGENT wait 1000

# ============ 点发送按钮 ============
# 首页输入框右下角发送按钮, 通常是 button 含 svg arrow 图标 + textContent 为空 / "Send"
$AGENT eval "(() => {
  const btn = [...document.querySelectorAll('button')]
    .filter(b => b.offsetParent && b.getBoundingClientRect().bottom > window.innerHeight - 200)
    .find(b => b.querySelector('svg') && (!b.textContent.trim() || /Send|发送/.test(b.textContent)));
  if (!btn) return 'no-send';
  btn.click();
  return 'sent';
})()"
# !!! 这一帧是 OBS 计时器启动的对齐点 !!!
$AGENT wait 4000

# ============ §7.1.0 综合信号轮询: 等任务完成 ============
for i in $(seq 1 40); do
  sleep 5
  STATE=$($AGENT eval "(() => {
    const txt = document.body.innerText;
    const tail = txt.slice(-2000);
    return JSON.stringify({
      done: tail.includes('任务完成') || tail.includes('Task completed'),
      web_search_seen: txt.includes('Searching') || txt.includes('reddit.com')
    });
  })()")
  echo "[$i] $STATE"
  echo "$STATE" | python3 -c "
import json,sys
s=sys.stdin.read()
try:
  o=json.loads(json.loads(s))
  exit(0 if o['done'] else 1)
except: exit(1)" && { echo '[OK] 任务完成'; break; }
done

# ============ 滚动展示 60 行种子词表 (终帧用) ============
$AGENT eval "document.querySelectorAll('.overflow-auto')[1].scrollTop = 0"
$AGENT wait 1500
$AGENT eval "(() => {
  const c = document.querySelectorAll('.overflow-auto')[1];
  c.scrollTop = c.scrollHeight;
  return c.scrollTop;
})()"
$AGENT wait 2500

echo "[DONE] R-1 录制完成,可停止 OBS"
echo "taskId: $($AGENT get url)"
```
